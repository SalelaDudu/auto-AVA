// ==UserScript==
// @name         Auto AVA
// @namespace    http://tampermonkey.net/
// @version      1.0
// @description  Extrai questões (com opções) em TXT e preenche respostas automaticamente.
// @author       Salela + Gemini
// @match        https://ava3.cefor.ifes.edu.br/mod/quiz/attempt.php*
// @match        https://ava3.cefor.ifes.edu.br/mod/quiz/review.php*
// @grant        none
// ==/UserScript==

(function() {
    'use strict';

    // Variáveis de controle de estado
    const isExtracting = localStorage.getItem('robo_extrator_ativo') === 'true';
    const isAnswering = localStorage.getItem('robo_respondedor_ativo') === 'true';

    // Roteamento de ações automáticas (silencioso, sem barra no topo)
    if (isExtracting) {
        processarPaginaAtualExtrator();
    } else if (isAnswering) {
        esperarEditorEPreencher();
    } else {
        criarInterface();
    }

    // ==========================================
    // 1. INTERFACE DO USUÁRIO (BOTÕES E MODAL)
    // ==========================================
    function criarInterface() {
        const container = document.createElement('div');
        container.style = 'position: fixed; bottom: 20px; right: 20px; z-index: 999999; display: flex; flex-direction: column; gap: 10px;';

        // Botão Extrator
        const btnExtrair = document.createElement('button');
        btnExtrair.innerText = 'Extrair Questoes (TXT)';
        btnExtrair.style = 'padding: 12px; background-color: #007bff; color: #fff; border: none; border-radius: 5px; cursor: pointer; box-shadow: 0px 4px 6px rgba(0,0,0,0.3); font-weight: bold; transition: background-color 0.3s;';
        btnExtrair.addEventListener('mouseenter', () => btnExtrair.style.backgroundColor = '#0056b3');
        btnExtrair.addEventListener('mouseleave', () => btnExtrair.style.backgroundColor = '#007bff');

        // Botão Respondedor
        const btnResponder = document.createElement('button');
        btnResponder.innerText = 'Responder Automatico';
        btnResponder.style = 'padding: 12px; background-color: #28a745; color: #fff; border: none; border-radius: 5px; cursor: pointer; box-shadow: 0px 4px 6px rgba(0,0,0,0.3); font-weight: bold; transition: background-color 0.3s;';
        btnResponder.addEventListener('mouseenter', () => btnResponder.style.backgroundColor = '#218838');
        btnResponder.addEventListener('mouseleave', () => btnResponder.style.backgroundColor = '#28a745');

        container.appendChild(btnExtrair);
        container.appendChild(btnResponder);
        document.body.appendChild(container);

        // Modal do Respondedor
        const modal = document.createElement('div');
        modal.style = 'display: none; position: fixed; top: 0; left: 0; width: 100%; height: 100%; background: rgba(0,0,0,0.7); z-index: 1000000; justify-content: center; align-items: center;';

        const modalContent = document.createElement('div');
        modalContent.style = 'background: #fff; padding: 25px; border-radius: 8px; width: 80%; max-width: 650px; display: flex; flex-direction: column; gap: 15px; box-shadow: 0px 10px 30px rgba(0,0,0,0.5);';

        const label = document.createElement('h3');
        label.innerText = 'Cole suas respostas abaixo:';
        label.style.margin = '0';
        label.style.fontFamily = 'Arial, sans-serif';

        const textarea = document.createElement('textarea');
        textarea.style = 'width: 100%; height: 350px; padding: 10px; font-family: monospace; resize: vertical; border: 1px solid #ccc; border-radius: 4px;';
        textarea.placeholder = "Padrao esperado:\n\n1) a\n\n29) blablalba...";

        const btnContainer = document.createElement('div');
        btnContainer.style = 'display: flex; justify-content: flex-end; gap: 10px;';

        const btnCancelar = document.createElement('button');
        btnCancelar.innerText = 'Cancelar';
        btnCancelar.style = 'padding: 10px 20px; cursor: pointer; border: none; border-radius: 5px; background: #dc3545; color: white; font-weight: bold;';

        const btnIniciar = document.createElement('button');
        btnIniciar.innerText = 'Iniciar Preenchimento';
        btnIniciar.style = 'padding: 10px 20px; cursor: pointer; border: none; border-radius: 5px; background: #007bff; color: white; font-weight: bold;';

        btnContainer.appendChild(btnCancelar);
        btnContainer.appendChild(btnIniciar);
        modalContent.appendChild(label);
        modalContent.appendChild(textarea);
        modalContent.appendChild(btnContainer);
        modal.appendChild(modalContent);
        document.body.appendChild(modal);

        // Eventos da UI
        btnExtrair.addEventListener('click', iniciarExtrator);
        btnResponder.addEventListener('click', () => { modal.style.display = 'flex'; });
        btnCancelar.addEventListener('click', () => { modal.style.display = 'none'; });
        btnIniciar.addEventListener('click', () => iniciarRespondedor(textarea.value));
    }

    // ==========================================
    // 2. LÓGICA DO EXTRATOR
    // ==========================================
    function iniciarExtrator() {
        if (confirm('O script vai navegar automaticamente por todas as paginas para copiar as questoes. Nao clique em nada. Deseja comecar?')) {
            localStorage.setItem('robo_extrator_ativo', 'true');
            localStorage.setItem('robo_extrator_questoes', JSON.stringify([]));
            irParaPagina1OuProcessar(processarPaginaAtualExtrator);
        }
    }

    function processarPaginaAtualExtrator() {
        setTimeout(() => {
            let questoesSalvas = JSON.parse(localStorage.getItem('robo_extrator_questoes') || '[]');
            const questionNodes = document.querySelectorAll('.que');

            questionNodes.forEach(qNode => {
                const qNoElement = qNode.querySelector('.qno');
                const qTextElement = qNode.querySelector('.qtext');

                if (qNoElement && qTextElement) {
                    let textoCompleto = qTextElement.innerText.trim();

                    // Lógica para extrair as opções de múltipla escolha
                    const options = qNode.querySelectorAll('.answer [data-region="answer-label"], .answer label');
                    if (options.length > 0) {
                        textoCompleto += '\n'; // Espaçamento entre enunciado e opções
                        options.forEach(opt => {
                            // Limpa quebras de linha sujas no meio da alternativa
                            textoCompleto += '\n' + opt.innerText.trim().replace(/\s+/g, ' ');
                        });
                    }

                    questoesSalvas.push({
                        numero: parseInt(qNoElement.innerText.trim(), 10),
                        texto: textoCompleto
                    });
                }
            });

            // Remove duplicatas
            const mapUnique = new Map();
            questoesSalvas.forEach(q => mapUnique.set(q.numero, q));
            questoesSalvas = Array.from(mapUnique.values());

            localStorage.setItem('robo_extrator_questoes', JSON.stringify(questoesSalvas));

            // Navegação
            const currentPageBtn = document.querySelector('.qnbutton.thispage');
            if (currentPageBtn) {
                const currentPgIndex = parseInt(currentPageBtn.getAttribute('data-quiz-page'), 10);
                const nextPgBtn = document.querySelector(`.qnbutton[data-quiz-page="${currentPgIndex + 1}"]`);

                if (nextPgBtn) {
                    window.location.href = nextPgBtn.href.split('#')[0];
                } else {
                    finalizarExtracao(questoesSalvas);
                }
            } else {
                finalizarExtracao(questoesSalvas);
            }
        }, 1000);
    }

    function finalizarExtracao(questoes) {
        localStorage.removeItem('robo_extrator_ativo');
        localStorage.removeItem('robo_extrator_questoes');

        questoes.sort((a, b) => a.numero - b.numero);

        let textoFinal = '';
        questoes.forEach(q => {
            textoFinal += `${q.numero}) ${q.texto}\n\n`;
        });

        if (questoes.length > 0) {
            baixarTXT(textoFinal, 'questoes_questionario.txt');
            alert(`Processo concluido. Extracao finalizada com ${questoes.length} questoes.`);
        } else {
            alert('Aviso: Nenhuma questao foi encontrada.');
        }

        location.reload();
    }

    // ==========================================
    // 3. LÓGICA DO RESPONDEDOR
    // ==========================================
    function iniciarRespondedor(textoBase) {
        if (!textoBase.trim()) {
            return alert('Cole o texto com as respostas antes de iniciar.');
        }

        const regex = /(?:^|\n)\s*(\d+)\)\s*([\s\S]*?)(?=\n\s*\d+\)|$)/g;
        let match;
        const respostas = {};
        let count = 0;

        while ((match = regex.exec(textoBase)) !== null) {
            respostas[match[1]] = match[2].trim();
            count++;
        }

        if (count === 0) {
            return alert('Falha ao identificar questoes. Use o padrao "1) resposta".');
        }

        if (confirm(`Foram identificadas ${count} respostas. O preenchimento iniciara automaticamente. Deseja prosseguir?`)) {
            localStorage.setItem('robo_respondedor_ativo', 'true');
            localStorage.setItem('robo_respostas', JSON.stringify(respostas));
            irParaPagina1OuProcessar(esperarEditorEPreencher);
        }
    }

    function esperarEditorEPreencher() {
        let tentativas = 0;
        const intervalo = setInterval(() => {
            tentativas++;
            // Verifica se as instâncias do TinyMCE existem ou se deu timeout (5s)
            if ((typeof tinymce !== 'undefined' && tinymce.editors && tinymce.editors.length > 0) || tentativas > 10) {
                clearInterval(intervalo);
                preencherRespostasEAvancar();
            }
        }, 500);
    }

    function preencherRespostasEAvancar() {
        const respostas = JSON.parse(localStorage.getItem('robo_respostas') || '{}');
        const questionNodes = document.querySelectorAll('.que');

        questionNodes.forEach(qNode => {
            const qNoElement = qNode.querySelector('.qno');
            if (!qNoElement) return;

            const qNo = qNoElement.innerText.trim();

            if (respostas[qNo]) {
                const textoResposta = respostas[qNo];

                // Elementos de Dissertação e Múltipla Escolha
                const textarea = qNode.querySelector('textarea[id$="_answer_id"], textarea.form-control');
                const radios = qNode.querySelectorAll('input[type="radio"]');

                // Lógica de Preenchimento para Dissertativas
                if (textarea) {
                    const id = textarea.id;
                    if (typeof tinymce !== 'undefined' && tinymce.get(id)) {
                        tinymce.get(id).setContent(textoResposta);
                    } else {
                        textarea.value = textoResposta;
                    }
                }
                // Lógica de Preenchimento para Múltipla Escolha
                else if (radios.length > 0) {
                    const targetLetter = textoResposta.toLowerCase();
                    let matched = false;

                    radios.forEach(radio => {
                        if (matched) return;
                        let labelText = '';

                        // Tenta encontrar o texto da alternativa pelo Moodle padrão (label)
                        const labelEl = qNode.querySelector(`label[for="${radio.id}"]`);
                        if (labelEl) {
                            labelText = labelEl.innerText.trim().toLowerCase();
                        } else {
                            // Tenta encontrar o texto da alternativa pelo tema do Cefor (aria-labelledby)
                            const ariaId = radio.getAttribute('aria-labelledby');
                            if (ariaId) {
                                const ariaEl = document.getElementById(ariaId);
                                if (ariaEl) labelText = ariaEl.innerText.trim().toLowerCase();
                            }
                        }

                        if (labelText) {
                            // Clica se começar com "a.", "a)" ou se for exatamente "a"
                            if (labelText.startsWith(targetLetter + '.') ||
                                labelText.startsWith(targetLetter + ')') ||
                                labelText === targetLetter) {
                                radio.click();
                                matched = true;
                            }
                        }
                    });
                }
            }
        });

        // Espera renderizar os cliques/textos e avança
        setTimeout(() => {
            const btnNext = document.querySelector('input[name="next"], button.mod_quiz-next-nav');
            if (btnNext) {
                btnNext.click();
            } else {
                finalizarPreenchimento();
            }
        }, 1500);
    }

    function finalizarPreenchimento() {
        localStorage.removeItem('robo_respondedor_ativo');
        localStorage.removeItem('robo_respostas');

        const painelFinal = document.createElement('div');
        painelFinal.style = 'position: fixed; top: 50%; left: 50%; transform: translate(-50%, -50%); background: #28a745; color: #fff; padding: 30px; z-index: 9999999; font-weight: bold; font-size: 20px; border-radius: 10px; box-shadow: 0 4px 15px rgba(0,0,0,0.5); text-align: center;';
        painelFinal.innerHTML = 'Processo Finalizado!<br><br><span style="font-size:15px; font-weight:normal;">As paginas foram percorridas com sucesso.<br>Confira na revisao se as questoes estao respondidas.</span><br><br><button onclick="location.reload()" style="margin-top:20px; padding:10px 25px; border:none; border-radius:5px; background:#fff; color:#28a745; font-weight:bold; cursor:pointer;">Entendi</button>';

        document.body.appendChild(painelFinal);
    }

    // ==========================================
    // 4. FUNÇÕES UTILITÁRIAS
    // ==========================================
    function baixarTXT(conteudo, nomeArquivo) {
        const blob = new Blob([conteudo], { type: 'text/plain;charset=utf-8' });
        const link = document.createElement('a');
        link.href = URL.createObjectURL(blob);
        link.download = nomeArquivo;
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    }

    function irParaPagina1OuProcessar(funcaoProcessamento) {
        const btnPage1 = document.querySelector('.qnbutton[data-quiz-page="0"]');
        if (btnPage1 && window.location.href.split('#')[0] !== btnPage1.href.split('#')[0]) {
            window.location.href = btnPage1.href.split('#')[0];
        } else {
            funcaoProcessamento === esperarEditorEPreencher ? location.reload() : funcaoProcessamento();
        }
    }

})();
